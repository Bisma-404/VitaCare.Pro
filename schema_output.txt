-- ===================================================================
-- HOSPITAL MANAGEMENT + MULTI-DISEASE DETECTION SYSTEM
-- Complete MySQL Database Schema
-- ===================================================================
-- Database: hospital_management_db (Updated to match app.py)
-- Engine: MySQL 8.0+
-- Includes: Tables, Foreign Keys, Constraints, Indexes, Sample Data
-- ===================================================================

-- Create Database
CREATE DATABASE IF NOT EXISTS hospital_management_db;
USE hospital_management_db;

-- ===================================================================
-- 1. USERS TABLE (Base table for all users: staff and patients)
-- ===================================================================
CREATE TABLE users (
    id INT PRIMARY KEY AUTO_INCREMENT,
    username VARCHAR(50) UNIQUE NOT NULL,
    password VARCHAR(100) NOT NULL,
    role ENUM('ADMIN', 'DOCTOR', 'LAB_TECH', 'PATIENT') NOT NULL,
    name VARCHAR(100) NOT NULL,
    email VARCHAR(100),
    phone VARCHAR(15),
    status ENUM('ACTIVE', 'INACTIVE', 'SUSPENDED') DEFAULT 'ACTIVE',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    last_updated DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_username (username),
    INDEX idx_role (role),
    INDEX idx_email (email),
    INDEX idx_status (status)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ===================================================================
-- 2. PATIENT PROFILES TABLE
-- ===================================================================
CREATE TABLE patient_profiles (
    id INT PRIMARY KEY AUTO_INCREMENT,
    patient_id INT NOT NULL UNIQUE,
    date_of_birth DATE,
    gender ENUM('MALE', 'FEMALE', 'OTHER'),
    blood_group VARCHAR(10),
    address TEXT,
    emergency_contact VARCHAR(100),
    emergency_phone VARCHAR(15),
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (patient_id) REFERENCES users(id) ON DELETE CASCADE,
    INDEX idx_patient_id (patient_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ===================================================================
-- 3. PATIENT REPORTS TABLE
-- ===================================================================
CREATE TABLE patient_reports (
    id INT PRIMARY KEY AUTO_INCREMENT,
    patient_id INT NOT NULL,
    staff_id INT NOT NULL,
    report_type ENUM('GENERAL', 'DIABETES', 'HEART', 'BREAST_CANCER', 'LAB_REPORT') DEFAULT 'GENERAL',
    uploaded_file VARCHAR(255),
    notes TEXT,
    status ENUM('PENDING', 'COMPLETED', 'ARCHIVED') DEFAULT 'PENDING',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (patient_id) REFERENCES users(id) ON DELETE CASCADE,
    FOREIGN KEY (staff_id) REFERENCES users(id) ON DELETE RESTRICT,
    INDEX idx_patient_id (patient_id),
    INDEX idx_staff_id (staff_id),
    INDEX idx_report_type (report_type),
    INDEX idx_created_at (created_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ===================================================================
-- 4. REPORT TESTS TABLE (Test results from reports)
-- ===================================================================
CREATE TABLE report_tests (
    id INT PRIMARY KEY AUTO_INCREMENT,
    report_id INT NOT NULL,
    test_name VARCHAR(100) NOT NULL,
    test_value DECIMAL(10,2) NOT NULL,
    unit VARCHAR(20),
    normal_range VARCHAR(50),
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (report_id) REFERENCES patient_reports(id) ON DELETE CASCADE,
    INDEX idx_report_id (report_id),
    INDEX idx_test_name (test_name)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ===================================================================
-- 5. DISEASE MODELS TABLE (Available disease models)
-- ===================================================================
CREATE TABLE disease_models (
    id INT PRIMARY KEY AUTO_INCREMENT,
    disease_name VARCHAR(100) NOT NULL UNIQUE,
    disease_code VARCHAR(50) NOT NULL UNIQUE,
    description TEXT,
    model_file_path VARCHAR(255),
    is_active BOOLEAN DEFAULT TRUE,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_disease_code (disease_code),
    INDEX idx_is_active (is_active)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ===================================================================
-- 6. DISEASE THRESHOLDS TABLE (Thresholds for DSA analysis)
-- ===================================================================
CREATE TABLE disease_thresholds (
    id INT PRIMARY KEY AUTO_INCREMENT,
    disease_id INT NOT NULL,
    parameter_name VARCHAR(100) NOT NULL,
    min_value DECIMAL(10,2),
    max_value DECIMAL(10,2),
    unit VARCHAR(20),
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (disease_id) REFERENCES disease_models(id) ON DELETE CASCADE,
    UNIQUE KEY unique_disease_parameter (disease_id, parameter_name),
    INDEX idx_disease_id (disease_id),
    INDEX idx_parameter_name (parameter_name)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ===================================================================
-- 7. SYMPTOMS TABLE
-- ===================================================================
CREATE TABLE symptoms (
    id INT PRIMARY KEY AUTO_INCREMENT,
    symptom_name VARCHAR(100) NOT NULL UNIQUE,
    description TEXT,
    category VARCHAR(50),
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_symptom_name (symptom_name),
    INDEX idx_category (category)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ===================================================================
-- 8. SYMPTOM-DISEASE MAPPING TABLE (Graph relationships)
-- ===================================================================
CREATE TABLE symptom_disease_mapping (
    id INT PRIMARY KEY AUTO_INCREMENT,
    symptom_id INT NOT NULL,
    disease_id INT NOT NULL,
    correlation_strength DECIMAL(3,2) DEFAULT 1.0,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (symptom_id) REFERENCES symptoms(id) ON DELETE CASCADE,
    FOREIGN KEY (disease_id) REFERENCES disease_models(id) ON DELETE CASCADE,
    UNIQUE KEY unique_symptom_disease (symptom_id, disease_id),
    INDEX idx_symptom_id (symptom_id),
    INDEX idx_disease_id (disease_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ===================================================================
-- 9. PATIENT SYMPTOMS TABLE (Symptoms reported by patients)
-- ===================================================================
CREATE TABLE patient_symptoms (
    id INT PRIMARY KEY AUTO_INCREMENT,
    report_id INT NOT NULL,
    symptom_id INT NOT NULL,
    severity ENUM('MILD', 'MODERATE', 'SEVERE') DEFAULT 'MODERATE',
    notes TEXT,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (report_id) REFERENCES patient_reports(id) ON DELETE CASCADE,
    FOREIGN KEY (symptom_id) REFERENCES symptoms(id) ON DELETE CASCADE,
    INDEX idx_report_id (report_id),
    INDEX idx_symptom_id (symptom_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ===================================================================
-- 10. PREDICTIONS TABLE (Main predictions table)
-- ===================================================================
CREATE TABLE predictions (
    id INT PRIMARY KEY AUTO_INCREMENT,
    report_id INT NOT NULL,
    disease_id INT NOT NULL,
    prediction_result INT NOT NULL COMMENT '0=Low Risk, 1=High Risk',
    confidence_score DECIMAL(5,2) NOT NULL COMMENT '0-100',
    method ENUM('DSA', 'ML', 'HYBRID') DEFAULT 'DSA',
    dsa_result_id INT,
    ml_result_id INT,
    show_in_patient_panel BOOLEAN DEFAULT FALSE COMMENT 'Flag to show prediction in patient dashboard',
    risk_score DECIMAL(5,2) COMMENT 'Calculated risk score',
    severity_label VARCHAR(50) COMMENT 'Risk severity: Low, Medium, High',
    severity_reason TEXT COMMENT 'Explanation of severity classification',
    threshold_violations JSON COMMENT 'Parameters exceeding normal thresholds',
    symptom_matches INT DEFAULT 0 COMMENT 'Number of symptoms matched',
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (report_id) REFERENCES patient_reports(id) ON DELETE CASCADE,
    FOREIGN KEY (disease_id) REFERENCES disease_models(id) ON DELETE CASCADE,
    INDEX idx_report_id (report_id),
    INDEX idx_disease_id (disease_id),
    INDEX idx_prediction_result (prediction_result),
    INDEX idx_confidence_score (confidence_score),
    INDEX idx_created_at (created_at),
    INDEX idx_show_in_patient_panel (show_in_patient_panel)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ===================================================================
-- 11. DSA RESULTS TABLE (Detailed DSA analysis results)
-- ===================================================================
CREATE TABLE dsa_results (
    id INT PRIMARY KEY AUTO_INCREMENT,
    prediction_id INT NOT NULL,
    threshold_violations JSON,
    symptom_matches INT DEFAULT 0,
    risk_score DECIMAL(5,2) NOT NULL,
    decision_rules_applied TEXT,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (prediction_id) REFERENCES predictions(id) ON DELETE CASCADE,
    INDEX idx_prediction_id (prediction_id),
    INDEX idx_risk_score (risk_score)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ===================================================================
-- 12. ML PREDICTION HISTORY TABLE (ML model predictions)
-- ===================================================================
CREATE TABLE ml_prediction_history (
    id INT PRIMARY KEY AUTO_INCREMENT,
    prediction_id INT NOT NULL,
    model_version VARCHAR(50),
    features_used JSON,
    prediction_probability DECIMAL(5,4),
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (prediction_id) REFERENCES predictions(id) ON DELETE CASCADE,
    INDEX idx_prediction_id (prediction_id),
    INDEX idx_created_at (created_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ===================================================================
-- 13. STAFF ACTIVITY LOG TABLE
-- ===================================================================
CREATE TABLE staff_activity_log (
    id INT PRIMARY KEY AUTO_INCREMENT,
    staff_id INT NOT NULL,
    action_type VARCHAR(50) NOT NULL,
    description TEXT,
    related_patient_id INT,
    related_report_id INT,
    ip_address VARCHAR(45),
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (staff_id) REFERENCES users(id) ON DELETE CASCADE,
    FOREIGN KEY (related_patient_id) REFERENCES users(id) ON DELETE SET NULL,
    FOREIGN KEY (related_report_id) REFERENCES patient_reports(id) ON DELETE SET NULL,
    INDEX idx_staff_id (staff_id),
    INDEX idx_action_type (action_type),
    INDEX idx_created_at (created_at)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ===================================================================
-- 14. LOGIN HISTORY TABLE
-- ===================================================================
CREATE TABLE login_history (
    id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL,
    ip_address VARCHAR(45),
    user_agent VARCHAR(255),
    success BOOLEAN DEFAULT TRUE,
    login_time DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    INDEX idx_user_id (user_id),
    INDEX idx_success (success),
    INDEX idx_login_time (login_time)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
